# ============================================================
# Dépendances Jetson Nano 2 Go — JetPack 4.6.x
# (Ubuntu 18.04, Python 3.6, CUDA 10.2, TensorRT 8.2)
# ============================================================
# Fourni par JetPack, NE PAS installer avec pip :
#   - OpenCV 4.1.1 (module cv2, compilé avec GStreamer et V4L2)
#   - TensorRT 8.2 (module tensorrt) et /usr/src/tensorrt/bin/trtexec
#   - CUDA 10.2 (/usr/local/cuda)
#
# INTERDIT sur la Jetson (2 Go de RAM partagés entre CPU et GPU) :
#   tensorflow, torch, ultralytics, onnxruntime, et surtout opencv-python
#   (il masquerait l'OpenCV de JetPack, qui est adapté à la carte).
#
# Préparation :
#   sudo apt install python3-pip python3-dev
#   python3 -m pip install --user --upgrade "pip<22"
#       (pip 21.3.1 = dernière version compatible Python 3.6 ; le pip d'origine
#        est trop ancien pour les paquets binaires aarch64 récents)
#   export PATH=/usr/local/cuda/bin:$PATH
#       (nvcc doit être trouvé pour compiler pycuda)
# Installation :
#   python3 -m pip install --user -r requirements-jetson.txt
# Avec un venv : python3 -m venv --system-site-packages venv
#   (sans cette option, cv2 et tensorrt de JetPack sont invisibles)
# ============================================================
